cmake_minimum_required(VERSION 3.16)
project(mediakit C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # static CRT: the exes run without the VC redistributable
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/MP)
endif()

option(MK_BUILD_EXAMPLES "build examples" ON)
option(MK_BUILD_TESTS "build tests" ON)
# x264 is GPL: ON makes every program linked with mediakit GPL. OFF keeps the build LGPL
# (H.264 encoding then uses the OS encoders only).
option(MK_WITH_X264 "build the x264 software H.264 encoder (GPL)" OFF)

# ---------------------------------------------------------------- codec (common to every OS)
include(codec/ffmpeg/ffmpeg.cmake)
if(MK_WITH_X264)
  include(codec/x264/x264.cmake)
  foreach(t mk_avutil mk_avcodec mk_avformat mk_swscale mk_swresample)
    target_compile_definitions(${t} PRIVATE MK_WITH_X264)
  endforeach()
  target_sources(mk_avcodec PRIVATE codec/ffmpeg/libavcodec/libx264.c)
  target_include_directories(mk_avcodec PRIVATE codec/x264)
  target_link_libraries(mk_avcodec PUBLIC mk_x264)
  message(STATUS "mediakit: x264 enabled - the result is GPL")
endif()

add_library(mk_codec STATIC codec/mk_codec.cpp)
target_include_directories(mk_codec PUBLIC common codec)
target_link_libraries(mk_codec PUBLIC mk_ffmpeg)
if(MSVC)
  target_compile_options(mk_codec PRIVATE /utf-8 /W3 /wd4244 /wd4267)
endif()

# ---------------------------------------------------------------- io (camera / microphone / speaker)
if(WIN32)
  add_library(mk_io STATIC io/win/mk_io_win.cpp)
  target_link_libraries(mk_io PUBLIC mfplat mfreadwrite mf mfuuid ole32 oleaut32 ksuser)
elseif(UNIX AND NOT APPLE AND NOT ANDROID)
  find_package(ALSA REQUIRED)
  add_library(mk_io STATIC io/linux/mk_io_linux.cpp)
  target_link_libraries(mk_io PUBLIC ALSA::ALSA Threads::Threads)
endif()
if(TARGET mk_io)
  target_include_directories(mk_io PUBLIC common io)
  if(MSVC)
    target_compile_options(mk_io PRIVATE /utf-8 /W3)
  endif()
endif()

# ---------------------------------------------------------------- pipeline (io + codec, no UI)
if(TARGET mk_io)
  add_library(mk_pipeline STATIC pipeline/mk_pipeline.cpp)
  target_include_directories(mk_pipeline PUBLIC pipeline)
  target_link_libraries(mk_pipeline PUBLIC mk_io mk_codec)
  if(MSVC)
    target_compile_options(mk_pipeline PRIVATE /utf-8 /W3 /wd4244 /wd4267)
  endif()
endif()

# ---------------------------------------------------------------- ui (per OS)
if(WIN32)
  add_library(mk_ui STATIC ui/win/mk_ui_win.cpp)
  target_link_libraries(mk_ui PUBLIC user32 gdi32 comdlg32 comctl32)
elseif(UNIX AND NOT APPLE AND NOT ANDROID)
  find_package(X11 REQUIRED)
  add_library(mk_ui STATIC ui/linux/mk_ui_x11.cpp)
  target_link_libraries(mk_ui PUBLIC X11::X11 Threads::Threads)
endif()
if(TARGET mk_ui)
  target_include_directories(mk_ui PUBLIC common ui)
  if(MSVC)
    target_compile_options(mk_ui PRIVATE /utf-8 /W3)
  endif()
endif()

# ---------------------------------------------------------------- programs
macro(mk_program name)
  add_executable(${name} ${ARGN})
  if(MSVC)
    target_compile_options(${name} PRIVATE /utf-8 /W3 /wd4244 /wd4267)
    target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
endmacro()

if(MK_BUILD_TESTS)
  mk_program(codec_test tests/codec_test.cpp)
  target_link_libraries(codec_test mk_codec)
  mk_program(mk_mux examples/mk_mux.cpp)
  target_link_libraries(mk_mux mk_codec)
endif()

if(MK_BUILD_EXAMPLES AND TARGET mk_io)
  mk_program(mk_devices examples/mk_devices.cpp)
  target_link_libraries(mk_devices mk_io)
  mk_program(mk_record examples/mk_record.cpp)
  target_link_libraries(mk_record mk_pipeline)
  mk_program(player_test tests/player_test.cpp)
  target_link_libraries(player_test mk_pipeline)
endif()
if(MK_BUILD_EXAMPLES AND TARGET mk_io AND TARGET mk_ui)
  mk_program(mk_play WIN32 examples/mk_play.cpp)
  target_link_libraries(mk_play mk_pipeline mk_ui)
  mk_program(mk_studio WIN32 examples/mk_studio.cpp)
  target_link_libraries(mk_studio mk_pipeline mk_ui)
  if(MSVC)  # GUI programs that still use main()
    set_target_properties(mk_play mk_studio PROPERTIES LINK_FLAGS "/ENTRY:mainCRTStartup")
  endif()
endif()
