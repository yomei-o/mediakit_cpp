cmake_minimum_required(VERSION 3.16)
project(mediakit C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # static CRT: the exes run without the VC redistributable
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/MP)
endif()

option(MK_BUILD_EXAMPLES "build examples" ON)
option(MK_BUILD_TESTS "build tests" ON)
# x264 is GPL: ON makes every program linked with mediakit GPL. OFF keeps the build LGPL
# (H.264 encoding then uses the OS encoders only).
option(MK_WITH_X264 "build the x264 software H.264 encoder (GPL)" OFF)
# OpenH264 is BSD 2-clause, so this one keeps the build LGPL. Slower and a little worse than
# x264 at the same bitrate, but it can be shipped in a closed product.
option(MK_WITH_OPENH264 "build the OpenH264 software H.264 encoder (BSD)" OFF)

# ---------------------------------------------------------------- codec (common to every OS)
include(codec/ffmpeg/ffmpeg.cmake)
if(MK_WITH_X264)
  include(codec/x264/x264.cmake)
  foreach(t mk_avutil mk_avcodec mk_avformat mk_swscale mk_swresample)
    target_compile_definitions(${t} PRIVATE MK_WITH_X264)
  endforeach()
  target_sources(mk_avcodec PRIVATE codec/ffmpeg/libavcodec/libx264.c)
  target_include_directories(mk_avcodec PRIVATE codec/x264)
  target_link_libraries(mk_avcodec PUBLIC mk_x264)
  message(STATUS "mediakit: x264 enabled - the result is GPL")
endif()
if(MK_WITH_OPENH264)
  include(codec/openh264/openh264.cmake)
  foreach(t mk_avutil mk_avcodec mk_avformat mk_swscale mk_swresample)
    target_compile_definitions(${t} PRIVATE MK_WITH_OPENH264)
  endforeach()
  target_sources(mk_avcodec PRIVATE
      codec/ffmpeg/libavcodec/libopenh264.c codec/ffmpeg/libavcodec/libopenh264enc.c)
  target_include_directories(mk_avcodec PRIVATE codec/openh264)
  target_link_libraries(mk_avcodec PUBLIC mk_openh264)
  message(STATUS "mediakit: OpenH264 enabled (BSD)")
endif()

add_library(mk_codec STATIC codec/mk_codec.cpp)
target_include_directories(mk_codec PUBLIC common codec)
target_link_libraries(mk_codec PUBLIC mk_ffmpeg)
if(EMSCRIPTEN)  # the browser's H.264 (WebCodecs): opening and flushing wait for its promises
  target_link_options(mk_codec PUBLIC -sASYNCIFY -sALLOW_MEMORY_GROWTH)
endif()
if(MSVC)
  target_compile_options(mk_codec PRIVATE /utf-8 /W3 /wd4244 /wd4267)
endif()

# ---------------------------------------------------------------- io (camera / microphone / speaker)
if(WIN32)
  add_library(mk_io STATIC io/win/mk_io_win.cpp)
  target_link_libraries(mk_io PUBLIC mfplat mfreadwrite mf mfuuid ole32 oleaut32 ksuser)
elseif(ANDROID)
  add_library(mk_io STATIC io/android/mk_io_android.cpp)
  target_link_libraries(mk_io PUBLIC camera2ndk mediandk aaudio android log)
elseif(APPLE)  # macOS and iOS share this one
  add_library(mk_io STATIC io/apple/mk_io_apple.mm)
  target_link_libraries(mk_io PUBLIC "-framework AVFoundation" "-framework AudioToolbox"
                                     "-framework CoreMedia" "-framework CoreVideo" "-framework Foundation")
  if(NOT IOS)
    target_link_libraries(mk_io PUBLIC "-framework CoreAudio")  # the device list; iOS has none
  endif()
elseif(EMSCRIPTEN)
  # browser: getUserMedia. open() waits for the browser's promise, so programs link with ASYNCIFY.
  add_library(mk_io STATIC io/wasm/mk_io_wasm.cpp)
  target_link_options(mk_io PUBLIC -sASYNCIFY -sALLOW_MEMORY_GROWTH)
elseif(UNIX AND NOT APPLE AND NOT ANDROID)
  find_package(ALSA REQUIRED)
  add_library(mk_io STATIC io/linux/mk_io_linux.cpp)
  target_link_libraries(mk_io PUBLIC ALSA::ALSA Threads::Threads)
endif()
if(TARGET mk_io)
  if(APPLE)
    target_compile_options(mk_io PRIVATE -fobjc-arc)
  endif()
  target_include_directories(mk_io PUBLIC common io)
  if(MSVC)
    target_compile_options(mk_io PRIVATE /utf-8 /W3)
  endif()
endif()

# ---------------------------------------------------------------- pipeline (io + codec, no UI)
if(TARGET mk_io)
  if(EMSCRIPTEN)  # no threads in the browser: the same classes, driven by JS timers
    add_library(mk_pipeline STATIC pipeline/mk_pipeline_wasm.cpp)
  else()
    add_library(mk_pipeline STATIC pipeline/mk_pipeline.cpp)
  endif()
  target_include_directories(mk_pipeline PUBLIC pipeline)
  target_link_libraries(mk_pipeline PUBLIC mk_io mk_codec)
  if(MSVC)
    target_compile_options(mk_pipeline PRIVATE /utf-8 /W3 /wd4244 /wd4267)
  endif()
endif()

# ---------------------------------------------------------------- ui (per OS)
if(WIN32)
  add_library(mk_ui STATIC ui/win/mk_ui_win.cpp)
  target_link_libraries(mk_ui PUBLIC user32 gdi32 comdlg32 comctl32)
elseif(ANDROID)
  # NativeActivity: the glue ships with the NDK, and this layer provides android_main()
  add_library(mk_ui STATIC ui/android/mk_ui_android.cpp
              ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
  target_include_directories(mk_ui PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
  target_link_libraries(mk_ui PUBLIC mk_codec android jnigraphics log)
elseif(IOS)
  add_library(mk_ui STATIC ui/ios/mk_ui_ios.mm)
  target_link_libraries(mk_ui PUBLIC "-framework UIKit" "-framework CoreGraphics" "-framework Foundation")
elseif(APPLE)
  add_library(mk_ui STATIC ui/mac/mk_ui_mac.mm)
  target_link_libraries(mk_ui PUBLIC "-framework Cocoa" "-framework UniformTypeIdentifiers")
elseif(EMSCRIPTEN)
  add_library(mk_ui STATIC ui/wasm/mk_ui_wasm.cpp)
  target_link_options(mk_ui PUBLIC -sASYNCIFY -sALLOW_MEMORY_GROWTH)
elseif(UNIX AND NOT APPLE AND NOT ANDROID)
  find_package(X11 REQUIRED)
  add_library(mk_ui STATIC ui/linux/mk_ui_x11.cpp)
  target_link_libraries(mk_ui PUBLIC X11::X11 Threads::Threads)
endif()
if(TARGET mk_ui)
  if(APPLE)
    target_compile_options(mk_ui PRIVATE -fobjc-arc)
  endif()
  target_include_directories(mk_ui PUBLIC common ui)
  if(MSVC)
    target_compile_options(mk_ui PRIVATE /utf-8 /W3)
  endif()
endif()

# ---------------------------------------------------------------- programs
macro(mk_program name)
  add_executable(${name} ${ARGN})
  if(MSVC)
    target_compile_options(${name} PRIVATE /utf-8 /W3 /wd4244 /wd4267)
    target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  if(APPLE AND NOT IOS)
    # TCC reads the usage strings out of the executable itself when there is no bundle
    target_link_options(${name} PRIVATE "SHELL:-sectcreate __TEXT __info_plist ${CMAKE_CURRENT_SOURCE_DIR}/tools/mac_info.plist")
    set_property(TARGET ${name} APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tools/mac_info.plist)
  endif()
endmacro()

  target_link_libraries(mk_mux mk_codec)
  if(EMSCRIPTEN)  # run with node; NODERAWFS gives them the real file system
    foreach(prog codec_test mk_mux)
      target_link_options(${prog} PRIVATE -sNODERAWFS=1 -sALLOW_MEMORY_GROWTH -sSTACK_SIZE=1048576)
    endforeach()
  endif()

if(MK_BUILD_EXAMPLES AND TARGET mk_io AND NOT EMSCRIPTEN)  # console programs that wait on threads
  mk_program(mk_devices examples/mk_devices.cpp)
  target_link_libraries(mk_devices mk_io)
  mk_program(mk_record examples/mk_record.cpp)
  target_link_libraries(mk_record mk_pipeline)
  mk_program(player_test tests/player_test.cpp)
  target_link_libraries(player_test mk_pipeline)
  mk_program(audio_loopback_test tests/audio_loopback_test.cpp)
  target_link_libraries(audio_loopback_test mk_io)
  mk_program(record_play_test tests/record_play_test.cpp)
  target_link_libraries(record_play_test mk_pipeline)
endif()
if(MK_BUILD_EXAMPLES AND EMSCRIPTEN)
  # browser versions of the tests (serve the build directory over http://localhost and open the .html):
  # audio_loopback_test speaker -> microphone, record_play_test Recorder -> MP4 -> Player
  mk_program(audio_loopback_test tests/audio_loopback_test.cpp)
  target_link_libraries(audio_loopback_test mk_io)
  mk_program(record_play_test tests/record_play_test.cpp)
  target_link_libraries(record_play_test mk_pipeline)
  set_target_properties(audio_loopback_test record_play_test PROPERTIES SUFFIX ".html")
endif()
if(MK_BUILD_EXAMPLES AND TARGET mk_io AND TARGET mk_ui)
  if(EMSCRIPTEN)
    # mk_vision.html + .js + .wasm; serve the build directory over http://localhost (camera access needs
    # a secure context) and open mk_vision.html. Browser frames are BGRA already, so no codec is linked.
    mk_program(mk_vision examples/mk_vision.cpp)
    target_link_libraries(mk_vision mk_io mk_ui)
    # mk_studio / mk_play: record and play with WebCodecs (recordings are downloaded; files to play are picked)
    mk_program(mk_studio examples/mk_studio.cpp)
    target_link_libraries(mk_studio mk_pipeline mk_ui)
    mk_program(mk_play examples/mk_play.cpp)
    target_link_libraries(mk_play mk_pipeline mk_ui)
    foreach(prog mk_vision mk_studio mk_play)
      set_target_properties(${prog} PROPERTIES SUFFIX ".html")
      target_link_options(${prog} PRIVATE --shell-file ${CMAKE_CURRENT_SOURCE_DIR}/tools/wasm_shell.html)
      set_property(TARGET ${prog} APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tools/wasm_shell.html)
    endforeach()
  elseif(ANDROID)
    # NativeActivity loads a shared library and calls android_main() in it (mk_ui provides one,
    # which calls the program's main()). tools/build_apk.sh wraps the .so into an APK.
    foreach(prog mk_play mk_studio)
      add_library(${prog} SHARED examples/${prog}.cpp)
      target_link_libraries(${prog} mk_pipeline mk_ui)
      target_link_options(${prog} PRIVATE "-u" "ANativeActivity_onCreate")
    endforeach()
  else()
    mk_program(mk_play WIN32 examples/mk_play.cpp)
    target_link_libraries(mk_play mk_pipeline mk_ui)
    mk_program(mk_studio WIN32 examples/mk_studio.cpp)
    target_link_libraries(mk_studio mk_pipeline mk_ui)
    mk_program(mk_vision WIN32 examples/mk_vision.cpp)
    target_link_libraries(mk_vision mk_io mk_codec mk_ui)
    if(MSVC)  # GUI programs that still use main()
      set_target_properties(mk_play mk_studio mk_vision PROPERTIES LINK_FLAGS "/ENTRY:mainCRTStartup")
    endif()
    if(IOS)  # .app bundles, so they can be installed on a device or a simulator
      foreach(prog mk_play mk_studio mk_vision)
        # an App ID takes only letters, digits, '-' and '.', so mk_studio becomes mkstudio.
        # Info.plist and the signed App ID must agree or installd rejects the app
        # with "The executable was signed with invalid entitlements".
        string(REPLACE "_" "" bundle_id ${prog})
        set_target_properties(${prog} PROPERTIES
            MACOSX_BUNDLE TRUE
            MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/tools/ios_info.plist.in
            MACOSX_BUNDLE_BUNDLE_NAME ${prog}
            MACOSX_BUNDLE_GUI_IDENTIFIER org.mediakit.${bundle_id}
            XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER org.mediakit.${bundle_id})
      endforeach()
    endif()
  endif()
endif()
